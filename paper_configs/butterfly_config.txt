# FineView butterfly scenes — 360° inward-facing object captures
# (8 calibrated cameras x 40 turntable positions = 320 views, white bg).
#
# dataset_type = llff refers to the DATA LOADER / file format only
# (poses_bounds.npy produced by fineview2nerf.py) — the data is the
# FineView butterfly dataset, not the LLFF forward-facing scenes.
#
# expname and datadir are overridden per species by run_butterfly_train.sh.

expname = butterfly
basedir = ./logs
datadir = ./data/nerf_butterfly/009-Neophasia_Menapia-001
dataset_type = llff

# Images are pre-resized by fineview2nerf.py (default width 800);
# factor = 1 uses them as-is (images_1 symlink avoids the minify step).
factor = 1

# Every 8th image held out for test — same protocol as 3DGS --eval.
llffhold = 8

# 360° object scene: NDC only works for forward-facing captures.
no_ndc = True
spherify = True
lindisp = True

# Images are masked to a white background; composite empty rays onto white
# so the network doesn't have to fill the volume with white density.
white_bkgd = True

# White butterfly on white background: ~80% of rays are pure background
# and the all-white "empty scene" is a strong local minimum (run 1+2 both
# collapsed into it: train PSNR pinned at ~16 dB = the all-white baseline).
# Follow the Blender/white_bkgd recipe: sample rays from one image per
# iteration (precrop only works on this path) and spend the first
# iterations on the image center, where the butterfly is, so density
# gradients outweigh the empty-scene pull.
no_batching = True
precrop_iters = 500
precrop_frac = 0.5

# FineView is a telephoto turntable capture: cameras sit ~20 object-radii
# away (FOV ~12°), so after spherify normalizes the camera sphere to
# radius 1 the butterfly is only ~0.1 units across. The default
# multires = 10 (max freq 2^9) gives ~8 positional-encoding periods
# across the object, while the images resolve it at ~400 px — the fine
# wing markings are unrepresentable. 14 (max freq 2^13) gives ~130
# periods, matching image resolution.
multires = 14

use_viewdirs = True
lrate_decay = 250
# raw_noise_std is the LLFF-scene floater regularizer; the Blender/white-bg
# recipe runs without it. With white_bkgd the noise also pushes raw sigma
# deeper into the dead-ReLU region, entrenching the empty-scene collapse.
raw_noise_std = 0.0

N_samples = 64
N_importance = 128

# N_rand 4096 in the paper; 1024 fits the RTX 4080 16 GB (see REPRODUCTION.md)
N_rand = 1024
N_iters = 200000
